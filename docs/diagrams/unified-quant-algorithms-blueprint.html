w_0 = 1.0
w_k = -w_{k-1} × (d - k + 1) / k   (|w_k| ≥ 10^-5)
P~_t = ∑_{k=0}^{K-1} w_k P_{t-k}
d* = min { d ∈ [0, 1] | p_ADF(P~_t(d)) < 0.05 }
